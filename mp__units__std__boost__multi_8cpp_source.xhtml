99[[maybe_unused]] const auto sample{[] -> int {
100 // The application owns the memory, here initialized in its representation.
101 storage a0({3, 1}, 0.);
102 a0.elements() = {3., 2., 1.};
103 state x0{std::mdspan{std::to_address(a0.data_elements()), column_extents{}}};
104
105 // Typed element access of the owned memory.
106 assert(x0.at<0>() == 3. * m);
107 assert(x0.at<1>() == 2. * m / s);
108 assert(x0.at<2>() == 1. * m / s2);
109
110 // Typed element assignment into the owned memory.
111 x0.at<1>(2.5 * m / s);
112 assert(std::ranges::equal(a0.elements(), std::array{3., 2.5, 1.}));
113
114 // Multiplication with a scalar factor, in place.
115 scale(3., x0);
116 assert(std::ranges::equal(a0.elements(), std::array{9., 7.5, 3.}));
117
118 storage a1({3, 1}, 0.);
119 state x1{std::mdspan{std::to_address(a1.data_elements()), column_extents{}}};
120
121 // Additions of two vectors of the same types.
122 add(x0, x0, x1);
123 assert(x1.at<0>() == 18. * m);
124 assert(x1.at<1>() == 15. * m / s);
125 assert(x1.at<2>() == 6. * m / s2);
126
127 storage a2({1, 3}, 0.);
128 a2.elements() = {3., 2., 1.};
129 state_transpose xt2{
130 std::mdspan{std::to_address(a2.data_elements()), row_extents{}}};
131
132 storage a3({3, 3}, 0.);
133 estimate_uncertainty p3{
134 std::mdspan{std::to_address(a3.data_elements()), matrix_extents{}}};
135
136 // Column-row matrix product, stored in row-major order.
137 matrix_product(x0, xt2, p3);
138 assert((p3.at<0, 0>() == 27. * m2));
139 assert((p3.at<1, 0>() == 22.5 * m2 / s));
140 assert((p3.at<2, 2>() == 3. * m2 / (s2 * s2)));
141 assert(std::ranges::equal(
142 a3.elements(), std::array{27., 18., 9., 22.5, 15., 7.5, 9., 6., 3.}));
143
144 return 0;
145}()};
