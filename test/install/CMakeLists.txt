# Verify the installed package is usable on its own: install the project into
# a fresh staging prefix, then configure and build a consumer project against
# it through the documented `find_package` interface. The installed public
# header includes every implementation file, so a header missing from the
# installed file set fails the consumer build.
set(PREFIX "${CMAKE_CURRENT_BINARY_DIR}/prefix")
set(CONSUMER "${CMAKE_CURRENT_BINARY_DIR}/consumer")

add_test(
  NAME typed_linear_algebra_install_clean
  COMMAND
    ${CMAKE_COMMAND}
    -E
    rm
    -r
    -f
    "${PREFIX}"
    "${CONSUMER}")
set_tests_properties(
  typed_linear_algebra_install_clean
  PROPERTIES FIXTURES_SETUP typed_linear_algebra_install_clean)

add_test(
  NAME typed_linear_algebra_install_install
  COMMAND
    ${CMAKE_COMMAND}
    --install
    "${CMAKE_BINARY_DIR}"
    --prefix
    "${PREFIX}"
    --config
    $<CONFIG>)
set_tests_properties(
  typed_linear_algebra_install_install
  PROPERTIES
    FIXTURES_REQUIRED typed_linear_algebra_install_clean FIXTURES_SETUP
    typed_linear_algebra_install_installed)

add_test(
  NAME typed_linear_algebra_install_find_package_pass
  COMMAND
    ${CMAKE_CTEST_COMMAND}
    --build-and-test
    "${CMAKE_CURRENT_SOURCE_DIR}/consumer"
    "${CONSUMER}"
    --build-generator
    "${CMAKE_GENERATOR}"
    --build-generator-platform
    "${CMAKE_GENERATOR_PLATFORM}"
    --build-generator-toolset
    "${CMAKE_GENERATOR_TOOLSET}"
    --build-makeprogram
    "${CMAKE_MAKE_PROGRAM}"
    --build-config
    $<CONFIG>
    --build-options
    "-DCMAKE_CXX_COMPILER=${CMAKE_CXX_COMPILER}"
    "-DCMAKE_PREFIX_PATH=${PREFIX}")
set_tests_properties(
  typed_linear_algebra_install_find_package_pass
  PROPERTIES FIXTURES_REQUIRED typed_linear_algebra_install_installed)
