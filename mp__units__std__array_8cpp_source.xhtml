90[[maybe_unused]] const auto sample{[] -> int {
91 // The application owns the memory, here initialized in its representation.
92 std::array a0{3., 2., 1.};
93 state x0{std::mdspan{a0.data(), column_extents{}}};
94
95 // Typed element access of the owned memory.
96 assert(x0.at<0>() == 3. * m);
97 assert(x0.at<1>() == 2. * m / s);
98 assert(x0.at<2>() == 1. * m / s2);
99
100 // Typed element assignment into the owned memory.
101 x0.at<1>(2.5 * m / s);
102 assert((a0 == std::array{3., 2.5, 1.}));
103
104 // Multiplication with a scalar factor, in place.
105 scale(3., x0);
106 assert((a0 == std::array{9., 7.5, 3.}));
107
108 std::array<representation, 3> a1{};
109 state x1{std::mdspan{a1.data(), column_extents{}}};
110
111 // Additions of two vectors of the same types.
112 add(x0, x0, x1);
113 assert(x1.at<0>() == 18. * m);
114 assert(x1.at<1>() == 15. * m / s);
115 assert(x1.at<2>() == 6. * m / s2);
116
117 std::array a2{3., 2., 1.};
118 state_transpose xt2{std::mdspan{a2.data(), row_extents{}}};
119
120 std::array<representation, 9> a3{};
121 estimate_uncertainty p3{std::mdspan{a3.data(), matrix_extents{}}};
122
123 // Column-row matrix product, stored in row-major order.
124 matrix_product(x0, xt2, p3);
125 assert((p3.at<0, 0>() == 27. * m2));
126 assert((p3.at<1, 0>() == 22.5 * m2 / s));
127 assert((p3.at<2, 2>() == 3. * m2 / (s2 * s2)));
128 assert((a3 == std::array{27., 18., 9., 22.5, 15., 7.5, 9., 6., 3.}));
129
130 return 0;
131}()};
