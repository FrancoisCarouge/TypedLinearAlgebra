112template <typename Lhs, typename Rhs>
113 requires fcarouge::armadillo::statically_sized<Lhs> &&
114 fcarouge::armadillo::statically_sized<Rhs>
115[[nodiscard]] inline auto operator==(const Lhs &lhs, const Rhs &rhs) -> bool {
116 static_assert(Lhs::n_rows == Rhs::n_rows && Lhs::n_cols == Rhs::n_cols,
117 "Equality requires matrices of the same shape.");
118
119 for (arma::uword k{0}; k < Lhs::n_rows * Lhs::n_cols; ++k) {
120 if (lhs[k] != rhs[k]) {
121 return false;
122 }
123 }
124
125 return true;
126}

193template <typename Type, typename Char>
194 requires fcarouge::armadillo::is_armadillo<Type> &&
195 fcarouge::armadillo::statically_sized<Type>
196struct std::formatter<Type, Char> {
197 static constexpr auto
198 parse(std::basic_format_parse_context<Char> &parse_context) {
199 return parse_context.begin();
200 }
201
202 template <typename OutputIterator>
203 constexpr auto
204 format(const Type &value,
205 std::basic_format_context<OutputIterator, Char> &format_context) const
206 -> OutputIterator {
207 auto output{format_context.out()};
208
217 auto write{[&output](const typename Type::elem_type &element) {
218 std::array<char, 64> buffer;
219 const auto end{
220 std::to_chars(buffer.data(), buffer.data() + buffer.size(), element)
221 .ptr};
222 output = std::copy(buffer.data(), end, output);
223 }};
224
225 if constexpr (Type::n_rows == 1 && Type::n_cols == 1) {
226 write(value.at(0, 0));
227 return output;
228 } else if constexpr (Type::n_rows == 1) {
229 *output++ = '[';
230 for (arma::uword j{0}; j < Type::n_cols; ++j) {
231 if (j != 0) {
232 *output++ = ',';
233 *output++ = ' ';
234 }
235 write(value.at(0, j));
236 }
237 *output++ = ']';
238 return output;
239 } else {
240 *output++ = '[';
241 for (arma::uword i{0}; i < Type::n_rows; ++i) {
242 if (i != 0) {
243 *output++ = ',';
244 *output++ = ' ';
245 }
246 *output++ = '[';
247 for (arma::uword j{0}; j < Type::n_cols; ++j) {
248 if (j != 0) {
249 *output++ = ',';
250 *output++ = ' ';
251 }
252 write(value.at(i, j));
253 }
254 *output++ = ']';
255 }
256 *output++ = ']';
257 return output;
258 }
259 }
260};
