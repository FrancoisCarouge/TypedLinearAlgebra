56[[maybe_unused]] const auto test{[] -> int {
57 // The typed linear algebra library's vector can be used as a representation
58 // for a quantity:
59 {
60 // A 3-component double-typed column vector, backed by an Eigen vector.
61 using vector3d = column_vector<double, double, double, double>;
62
63 // The homogeneous, double-typed vector satisfies mp-units's representation
64 // requirements to back up a vector quantity.
65 mp_units::quantity velocity{
66 mp_units::isq::velocity(vector3d{30, 40, 0} * km / h)};
67
68 // From there the framework treats the vector exactly like any other
69 // representation:
70 // * Unit conversion scales every component:
71 mp_units::quantity velocity_mps{velocity.in(m / s)};
72 // * vector quantity × scalar quantity:
73 mp_units::quantity displacement{velocity * (2 * h)};
74 // * Euclidean magnitude → scalar quantity:
75 mp_units::quantity speed{magnitude(velocity)};
76
77 assert(std::format("{}", velocity) == "[[30], [40], [0]] km/h");
78 assert(std::format("{}", velocity_mps) ==
79 "[[8.333333333333334], [11.11111111111111], [0]] m/s");
80 assert(std::format("{}", displacement) == "[[60], [80], [0]] km");
81 assert(std::format("{}", speed) == "50 km/h");
82 assert(std::format("{}", velocity.numerical_value_in(velocity.unit)) ==
83 "[[30], [40], [0]]");
84 }
85
86 // An mp-units vector quantity with strongly typed axes can be converted to a
87 // heterogeneous typed vector of quantities.
88 {
89 // A 3-component double-typed column vector, backed by an Eigen vector.
90 using vector3d = column_vector<double, double, double, double>;
91
92 // A velocity vector quantity with strongly typed axes backed by a
93 // 3-component double-typed column vector, backed by an Eigen vector.
94 mp_units::quantity velocity{flight_velocity(vector3d{30, 40, 0} * km / h)};
95
96 // A heterogeneous 3-component quantity-typed column vector, backed by an
97 // Eigen vector.
98 using velocities =
99 column_vector<double, mp_units::quantity<forward_velocity[km / h]>,
100 mp_units::quantity<lateral_velocity[km / h]>,
101 mp_units::quantity<vertical_velocity[km / h]>>;
102
103 // Convert the vector quantity to a heterogeneous typed column vector of
104 // quantities, out of the box.
105 velocities state{velocity};
106
107 // Note the difference in presentation of the vector state compared to the
108 // quantity vector:
109 assert(std::format("{}", state) == "[[30 km/h], [40 km/h], [0 km/h]]");
110 assert(std::format("{}", velocity) == "[[30], [40], [0]] km/h");
111
112 // The vector quantity has a magnitude, but not the heterogeneous typed
113 // vector of quantities, which doesn't have a meaning of a mathematical
114 // vector nor homogeneous units.
115 mp_units::quantity speed{magnitude(velocity)}; // 🗸
116 // quantity speed{magnitude(state)}; // 𐄂

118 assert(std::format("{}", speed) == "50 km/h");
119 }

121 // An mp-units vector quantity with strongly typed axes can be converted to a
122 // homogeneous typed vector of quantities given compatible quantity
123 // specifications.
124 {
125 // A 3-component double-typed column vector, backed by an Eigen vector.
126 using vector3d = column_vector<double, double, double, double>;
127
128 // A velocity vector quantity with strongly typed axes backed by a
129 // 3-component double-typed column vector, backed by an Eigen vector.
130 mp_units::quantity velocity{flight_velocity(vector3d{30, 40, 0} * km / h)};

132 assert(std::format("{}", get<0>(velocity)) == "30 km/h");
133 assert(std::format("{}", get<forward_velocity>(velocity)) == "30 km/h");
134
135 // The library also permits the conversion to a strongly typed column
136 // vector, backed by Eigen, with three components, each of type ISQ velocity
137 // quantity, if the conversion supports the quantity specific cast.
138 using velocities =
139 column_vector<double,
140 mp_units::quantity<mp_units::isq::velocity[km / h]>,
141 mp_units::quantity<mp_units::isq::velocity[km / h]>,
