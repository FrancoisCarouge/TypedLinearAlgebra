56// A vector quantity whose specification has no named-axis decomposition (no
57// `vector_components` specialization) but whose own representation is itself
58// tuple-like (e.g. a fcarouge typed vector). `not requires { ...::size; }`
59// mirrors mp-units's own `Decomposable` prerequisite check and is therefore
60// guaranteed mutually exclusive with it: mp-units's `std::tuple_size` and
61// `get<Idx>` specializations for `mp_units::quantity` require `Decomposable`,
62// which itself requires `vector_components<Spec>::size` to exist.
63template <auto Reference, typename Representation>
64concept undecomposed_tuple_like_quantity =
65 (not requires {
66 mp_units::vector_components<get_quantity_spec(Reference)>::size;
67 }) and requires(const Representation &representation) {
68 {
69 std::tuple_size<Representation>::value
70 } -> std::convertible_to<std::size_t>;
71 get<0>(representation);
72 };
73} // namespace fcarouge

75// Extend the standard tuple-like protocol (`std::tuple_size`,
76// `std::tuple_element`, and the ADL-found `get<Idx>` below) to vector
77// quantities left undecomposed by mp-units itself, forwarding to their own
78// tuple-like representation. This is what lets `fcarouge::typed_matrix`'s
79// core, vanilla-C++ `other_tuple_like_vector` converting constructor accept
80// such a quantity directly: `get<Idx>` is found purely through
81// argument-dependent lookup on `Representation` (which names `fcarouge`, and
82// the underlying matrix backend), so the core library never names any
83// mp-units-specific member.
84template <auto Reference, typename Representation>
85 requires fcarouge::undecomposed_tuple_like_quantity<Reference, Representation>
86struct std::tuple_size<mp_units::quantity<Reference, Representation>>
87 : std::tuple_size<Representation> {};

90 requires fcarouge::undecomposed_tuple_like_quantity<Reference,
91 Representation> and
92 (Index < std::tuple_size_v<Representation>)
93struct std::tuple_element<Index,
94 mp_units::quantity<Reference, Representation>> {
95 using type = mp_units::quantity<Reference,
96 std::tuple_element_t<Index, Representation>>;
97};

108// Teach the typed linear algebra library how to convert underlying scalar types
109// to and from mp-units' types. Support castable quantity specifications.
110template <typename To, mp_units::Quantity From>
111struct element_caster<To, From> {
112 [[nodiscard]] static constexpr auto operator()(mp_units::Quantity auto value)
113 -> To {
114 using representation = typename std::remove_cvref_t<From>::rep;
115
116 static_assert(std::same_as<representation, std::remove_cvref_t<To>>,
117 "The underlying storage type must be identical to the "
118 "quantity representation type to guarantee the conversion is "
119 "explicitly decided by the end-user.");
120
121 constexpr auto reference{std::remove_cvref_t<From>::reference};
122 constexpr auto specification{get_quantity_spec(reference)};
123 auto specified{mp_units::quantity_cast<specification>(value)};
124
125 return specified.numerical_value_in(specified.unit);
126 }
127};

233// True when at least one of the matrix's logical elements (`row index x
234// column index`, per element) is itself an mp-units quantity, as opposed to a
235// plain arithmetic representation (e.g. `double`). A homogeneous,
236// plain-representation typed vector or matrix (e.g. `column_vector<double,
237// double, double, double>`) is a legitimate mp-units Representation: this
238// is what backs `representation.cpp` and the first scenario of
239// `typed_vector_as_quantity.cpp`. A typed matrix whose elements are
240// themselves quantities (e.g. a per-axis heterogeneous velocity vector, or
241// any rank-0 "singleton" wrapping a single quantity) is a *result* of
242// decomposing a quantity, not a candidate storage for one; see
243// `disable_representation` below for why it must say so.
244template <typename Type>
245concept quantity_element_typed_matrix =
246 tla::same_as_typed_matrix<Type> and ([] {
247 bool result{false};
248
249 tla::for_constexpr<std::remove_cvref_t<Type>::rows>([&result](auto i) {
250 tla::for_constexpr<std::remove_cvref_t<Type>::columns>(
251 [&result, &i](auto j) {
252 result |= mp_units::Quantity<tla::element_at<Type, i, j>>;
253 });
254 });
255
256 return result;
257 }());
258} // namespace fcarouge

261// A typed_matrix is a composite facade over a backend matrix, never itself a
262// quantity's scalar/vector/tensor storage (its own elements, individually,
263// fill that role): opt it out of representation status exactly when its
264// elements are themselves quantities. Left unguarded, mp-units' generic
265// representation detection (RepresentationOf, consulted by quantity's own
266// operator==, operator+, and friends when overload resolution considers a
267// typed_matrix-of-quantities operand) would classify that matrix's character
268// by evaluating its operators; those operators are in turn defined in terms
269// of the matrix's own quantity elements, which recurses back into the same
270// RepresentationOf query already in flight for the matrix itself. That
271// self-reference is what surfaces as "satisfaction of constraint
272// 'RealScalar<T>' depends on itself". Declining representation status
273// upfront, the same guard mp-units applies to its own `quantity` (see
274// `disable_representation`'s default and `RepresentationBaseline`), stops
275// the character check, and therefore the cycle, before it starts.
276//
277// A typed_matrix of plain, non-quantity elements (e.g. a homogeneous
278// `column_vector<double, double, double, double>`) is left enabled: it is
279// exactly the kind of representation `RepresentationOf` is meant to detect,
280// and its element-wise operators only ever involve plain arithmetic, so no
281// cycle is possible.
282template <typename Matrix, typename RowIndexes, typename ColumnIndexes>
283inline constexpr bool disable_representation<
284 fcarouge::typed_matrix<Matrix, RowIndexes, ColumnIndexes>> =
285 fcarouge::quantity_element_typed_matrix<
286 fcarouge::typed_matrix<Matrix, RowIndexes, ColumnIndexes>>;
287} // namespace mp_units
