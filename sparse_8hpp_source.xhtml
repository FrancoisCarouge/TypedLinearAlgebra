136template <typename TypedMatrix, typename... Values>
137 requires(sizeof...(Values) == TypedMatrix::rows * TypedMatrix::columns) and
138 are_given<TypedMatrix, Values...>
139[[nodiscard]] constexpr auto make_matrix(const Values &...values)
140 -> TypedMatrix {
141 TypedMatrix result;
142
143 [&result, &values...]<std::size_t... Positions>(
144 std::index_sequence<Positions...>) -> void {
145 const auto write{[&result]<std::size_t Position>(
146 const auto &value) -> void {
147 if constexpr (not std::same_as<std::remove_cvref_t<decltype(value)>,
148 structural_zero_t>) {
149 static constexpr std::size_t row{Position / TypedMatrix::columns};
150 static constexpr std::size_t column{Position % TypedMatrix::columns};
151 static constexpr std::size_t index{
152 TypedMatrix::matrix::structure.index_of(row, column)};
153
154 using element = tla::element_at<TypedMatrix, row, column>;
155
156 // Not braces: a conversion to the element type may narrow.
157 const element converted(value);
158
159 result.data().values[index] =
160 cast<typename TypedMatrix::underlying, element>(converted);
161 }
162 }};
163
164 (write.template operator()<Positions>(values), ...);
165 }(std::index_sequence_for<Values...>{});
166
167 return result;
168}
