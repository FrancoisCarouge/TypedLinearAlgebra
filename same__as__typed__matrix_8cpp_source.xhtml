62 using m = matrix<double, 2, 2>;

73 // Negative, special case: fundamental and unrelated types have no nested
74 // `matrix`, `row_indexes`, `column_indexes` member types for the concept to
75 // probe. That probe is in the immediate context of the constraint, so the
76 // substitution failure makes the concept simply unsatisfied; it does not
77 // make the program ill-formed. `void` is included since it cannot have
78 // members at all.
79 static_assert(not same_as_typed_matrix<void>);
80 static_assert(not same_as_typed_matrix<int>);
81 static_assert(not same_as_typed_matrix<double>);
82 static_assert(not same_as_typed_matrix<std::string>);
83 static_assert(not same_as_typed_matrix<std::tuple<double>>);
84
85 // Negative, special case: an incomplete type also has no accessible nested
86 // types yet, and probing one still fails safely instead of hard-erroring on
87 // "invalid use of incomplete type".
88 {
89 struct incomplete;
90 static_assert(not same_as_typed_matrix<incomplete>);
91 }
92
93 // Negative: a pointer to a typed matrix is not itself a typed matrix.
94 // `remove_cvref_t` strips references and top-level cv-qualification, not
95 // indirection.
96 {
97 using m = matrix<double, 1, 1>;
98 static_assert(not same_as_typed_matrix<m *>);
99 static_assert(not same_as_typed_matrix<const m *>);
100 }
101
102 // Negative: the underlying backend matrix, composed into a typed matrix but
103 // not itself one, does not satisfy the concept.
104 static_assert(not same_as_typed_matrix<eigen::matrix<double, 2, 2>>);
105
106 // Negative, special case: the concept is not structural, duck typing. A
107 // type exposing the same `matrix`, `row_indexes`, and `column_indexes`
108 // member aliases as a typed matrix, but that is not literally that
109 // `typed_matrix` class template instantiation, does not satisfy the
110 // concept: `std::same_as` requires nominal type identity.
111 {
112 struct fake {
113 using matrix = eigen::matrix<double, 1, 1>;
114 using row_indexes = std::tuple<double>;
115 using column_indexes = std::tuple<double>;
116 };
117 static_assert(not same_as_typed_matrix<fake>);
118 }
119
120 // Negative, special case: a type publicly inheriting from a typed matrix
121 // inherits its `matrix`, `row_indexes`, `column_indexes` member aliases, so
122 // the concept reconstructs the *base* typed matrix from them, but
123 // `std::same_as` then compares the derived type itself against that
124 // reconstructed base. They are different types, so a derived type never
125 // satisfies `same_as_typed_matrix`, even though it "is a" typed matrix in
126 // the inheritance sense and adds no members of its own. One consequence:
127 // such a derived type instead satisfies `other`, so operators and
128 // constructors overloaded on `same_as_typed_matrix` versus `other` would
129 // treat it as "anything else", not as a typed matrix.
130 {
131 using base = matrix<double, 1, 1>;
132 struct derived : base {};
133 static_assert(same_as_typed_matrix<base>);
134 static_assert(not same_as_typed_matrix<derived>);
135 static_assert(other<derived>);
136 }
137
138 // Positive: `other` is exactly the negation of `same_as_typed_matrix`,
139 // practical for disambiguating overloads between a typed matrix and
140 // anything else.
141 static_assert(other<int>);
142 static_assert(not other<matrix<double, 1, 1>>);
