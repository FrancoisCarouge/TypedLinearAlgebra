70// The fcarouge::typed_matrix arithmetic operators return lazy
71// Armadillo expression templates; canonicalize an mp-units representation back
72// to the concrete, compile-time sized matrix type. The dimensions come from
73// the index tuples. The `typename Matrix::elem_type` requirement
74// is checked first and short-circuits the rest of the constraint.
75template <typename Matrix, typename RowIndexes, typename ColumnIndexes>
76 requires requires {
77 typename Matrix::elem_type;
78 } && std::derived_from<Matrix, arma::Base<typename Matrix::elem_type, Matrix>>
79struct mp_units::representation_canonical_type<
80 fcarouge::typed_matrix<Matrix, RowIndexes, ColumnIndexes>> {
81 using type = fcarouge::typed_matrix<
82 typename arma::Mat<typename Matrix::elem_type>::template fixed<
83 std::tuple_size_v<RowIndexes>, std::tuple_size_v<ColumnIndexes>>,
84 RowIndexes, ColumnIndexes>;
85};

87template <typename Matrix, typename RowIndexes, typename ColumnIndexes>
