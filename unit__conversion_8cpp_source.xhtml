70// A 3-component double-typed column vector.
71using vector3d = column_vector<double, double, double, double>;

73// The per-axis flight velocity elements, declared in kilometers per hour.
74using forward_kmph = mp_units::quantity<forward_velocity[km / h]>;
75using lateral_kmph = mp_units::quantity<lateral_velocity[km / h]>;
76using vertical_kmph = mp_units::quantity<vertical_velocity[km / h]>;
77using velocities =

80// The same axes, swapped.
81using swapped_velocities =

84// The per-axis flight velocity sources, in meters per second.
85using forward_mps = mp_units::quantity<forward_velocity[m / s]>;
86using lateral_mps = mp_units::quantity<lateral_velocity[m / s]>;
87using vertical_mps = mp_units::quantity<vertical_velocity[m / s]>;

89// A uniformly typed speed vector, declared in kilometers per hour.
90using speed_kmph = mp_units::quantity<speed[km / h]>;
91using speed_mps = mp_units::quantity<speed[m / s]>;
92using speeds = column_vector<double, speed_kmph, speed_kmph>;

94// The flight velocity components without their axes, declared in kilometers
95// per hour.
96using velocity_kmph = mp_units::quantity<velocity[km / h]>;
97using generic_velocities =
98 column_vector<double, velocity_kmph, velocity_kmph, velocity_kmph>;
99
100// The flight velocity vector quantities.
101using flight_velocity_kmph =
102 mp_units::quantity<flight_velocity[km / h], vector3d>;
103using flight_velocity_mps =
104 mp_units::quantity<flight_velocity[m / s], vector3d>;
105
106using velocity_mps = mp_units::quantity<velocity[m / s]>;
107
108// A tuple-like value implicitly converts when each of its elements implicitly
109// converts to the corresponding element type: same axes, any unit.
110static_assert(std::is_constructible_v<
111 velocities, std::tuple<forward_mps, lateral_mps, vertical_mps>>);
112static_assert(std::is_constructible_v<velocities, flight_velocity_mps>);
113static_assert(std::convertible_to<flight_velocity_mps, velocities>);
114static_assert(other_tuple_like_vector<
115 std::tuple<forward_mps, lateral_mps, vertical_mps>, velocities>);
116
117// The axes are not interchangeable, neither from a tuple nor from a vector
118// quantity decomposing in a different axes order.
119static_assert(
120 not std::is_constructible_v<
121 velocities, std::tuple<lateral_kmph, forward_kmph, vertical_kmph>>);
122static_assert(
123 not std::is_constructible_v<swapped_velocities, flight_velocity_kmph>);
124static_assert(
125 not other_tuple_like_vector<
126 std::tuple<lateral_kmph, forward_kmph, vertical_kmph>, velocities>);
127
128// An axis and its more general quantity only explicitly convert to each other,
129// so does a tuple-like value of them.
130static_assert(std::is_constructible_v<forward_kmph, velocity_mps>);
131static_assert(not std::is_convertible_v<velocity_mps, forward_kmph>);
132static_assert(
133 not tla::is_convertible_tuple_like<
134 std::tuple<velocity_mps, velocity_mps, velocity_mps>, velocities>());
135static_assert(
136 std::is_constructible_v<
137 velocities, std::tuple<velocity_mps, velocity_mps, velocity_mps>>);
138static_assert(
139 not std::convertible_to<
140 std::tuple<velocity_mps, velocity_mps, velocity_mps>, velocities>);
141static_assert(std::is_constructible_v<generic_velocities, flight_velocity_mps>);
142static_assert(not std::convertible_to<flight_velocity_mps, generic_velocities>);
143
144// Neither do unrelated quantities nor plain numbers.
145using length_m = mp_units::quantity<length[m]>;
146static_assert(not std::is_constructible_v<
147 velocities, std::tuple<length_m, length_m, length_m>>);
148static_assert(not std::is_constructible_v<velocities,
149 std::tuple<double, double, double>>);
150static_assert(not std::is_constructible_v<velocities, std::array<double, 3>>);
151
152// Elements convert to a typed vector as a tuple-like value of them would: of
153// the vector's size, implicitly when every element implicitly converts,
154// explicitly when some element only explicitly converts.
155static_assert(std::is_constructible_v<velocities, forward_mps, lateral_mps,
156 vertical_mps>);
157static_assert(
158 not std::is_constructible_v<velocities, forward_mps, lateral_mps>);
159static_assert(not std::is_constructible_v<velocities, forward_mps, lateral_mps,
160 vertical_mps, vertical_mps>);
161static_assert(not std::is_constructible_v<velocities, lateral_kmph,
162 forward_kmph, vertical_kmph>);
163static_assert(std::is_constructible_v<velocities, velocity_mps, velocity_mps,
164 velocity_mps>);
165static_assert(
166 not std::is_constructible_v<velocities, length_m, length_m, length_m>);
167static_assert(not std::is_constructible_v<velocities, double, double, double>);
