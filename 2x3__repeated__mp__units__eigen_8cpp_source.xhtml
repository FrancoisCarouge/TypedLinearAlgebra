41using representation = double;

43template <auto QuantityReference>
44using quantity = mp_units::quantity<QuantityReference, representation>;
45
46using mp_units::si::unit_symbols::kg;
47using mp_units::si::unit_symbols::m;
48using mp_units::si::unit_symbols::s;
49
50using length = quantity<mp_units::isq::length[m]>;
51using mass = quantity<mp_units::isq::mass[kg]>;
52using time = quantity<mp_units::isq::time[s]>;
53using length_time = decltype(1. * m * s);
54using mass_time = decltype(1. * kg * s);
55
56using rows = std::tuple<length, mass>;
57using columns = std::tuple<std::identity, std::identity, time>;
58using state = matrix<representation, rows, columns>;
59
60template <typename Matrix, typename Type>
61concept locatable = requires(Matrix value) { value.template at<Type>(); };
62
68[[maybe_unused]] const auto test{[] -> int {
69 // Row-major element types:
70 // (0,0) length (0,1) length (0,2) length*time
71 // (1,0) mass (1,1) mass (1,2) mass*time
72 state x;
73
74 x.at<0, 0>(1. * m);
75 x.at<0, 1>(2. * m);
76 x.at<0, 2>(3. * m * s);
77 x.at<1, 0>(4. * kg);
78 x.at<1, 1>(5. * kg);
79 x.at<1, 2>(6. * kg * s);
80
81 assert((x.at<length_time>() == 3. * m * s));
82 assert((x.at<mass_time>() == 6. * kg * s));
83 assert((x.at<length_time>() == x.at<0, 2>()));
84 assert((x.at<mass_time>() == x.at<1, 2>()));
85
86 static_assert(
87 std::same_as<decltype(x.at<length_time>()), decltype(x.at<0, 2>())>);
88 static_assert(
89 std::same_as<decltype(x.at<mass_time>()), decltype(x.at<1, 2>())>);
90 static_assert(not uniform_typed_matrix<state>);
91 static_assert(locatable<state, length_time>);
92 static_assert(locatable<state, mass_time>);
93 static_assert(not locatable<state, length>);
94 static_assert(not locatable<state, mass>);
95 static_assert(not locatable<state, time>);
96
97 x.at<1, 2>(9. * kg * s);
98
99 assert((x.at<mass_time>() == 9. * kg * s));
100
101 return 0;
102}()};
