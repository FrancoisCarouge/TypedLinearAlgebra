118// The call operators that compute a product, rather than name one, are

302template <std::size_t Columns>
303[[nodiscard]] constexpr auto
304storage_position(std::convertible_to<std::size_t> auto... indexes)
305 -> std::array<std::size_t, 2> {
306 if constexpr (sizeof...(indexes) == 2) {
307 return {static_cast<std::size_t>(indexes)...};
308 } else if constexpr (sizeof...(indexes) == 1) {
309 const std::size_t index{static_cast<std::size_t>(indexes)...};
310 return {index / Columns, index % Columns};
311 } else {
312 return {0, 0};
313 }
314}

328template <std::size_t Columns, typename Storage>
329[[nodiscard]] constexpr auto
330read_element(Storage &&storage,
331 std::convertible_to<std::size_t> auto... indexes)
332 -> decltype(auto) {
333 const auto [row, column]{storage_position<Columns>(indexes...)};
334
335 if constexpr (same_as_typed_matrix<Storage>) {
336 return read_element<Columns>(storage.data(), indexes...);
337 } else if constexpr (requires { access_element(storage, row, column); }) {
338 using storage_index = storage_index_t<std::remove_reference_t<Storage>>;
339
340 return access_element(storage, static_cast<storage_index>(row),
341 static_cast<storage_index>(column));
342 } else if constexpr (requires { storage.eval(); }) {
343 // `auto{...}` copies the element out of the local evaluation before it
344 // goes out of scope.
345 auto evaluated{storage.eval()};
346 return auto{read_element<Columns>(evaluated, indexes...)};
347 } else {
348 static_assert(sizeof(Storage) == 0,
349 "Element access is not supported for this linear algebra "
350 "backend.");
351 }
352}

367template <std::size_t Columns, auto... Indexes, typename Storage>
368[[nodiscard]] constexpr auto read_element_at(Storage &&storage)
369 -> decltype(auto) {
370 static constexpr std::array<std::size_t, 2> position{
371 storage_position<Columns>(Indexes...)};
372
373 using row = std::integral_constant<std::size_t, position[0]>;
374 using column = std::integral_constant<std::size_t, position[1]>;
375
376 if constexpr (same_as_typed_matrix<Storage>) {
377 return read_element_at<Columns, Indexes...>(storage.data());
378 } else if constexpr (requires { storage[row{}, column{}]; }) {
379 return storage[row{}, column{}];
380 } else {
381 return read_element<Columns>(storage, Indexes...);
382 }
383}

389template <std::size_t Columns, typename Storage, typename Value>
390constexpr void write_element(Storage &&storage, Value &&value,
391 std::convertible_to<std::size_t> auto... indexes) {
392 static_assert(std::is_lvalue_reference_v<decltype(read_element<Columns>(
393 storage, indexes...))>,
394 "Element write access requires a referenceable storage "
395 "element.");
396
397 read_element<Columns>(storage, indexes...) = std::forward<Value>(value);

428template <std::size_t Columns, auto... Indexes, typename Storage,
429 typename Value>
430constexpr void write_element_at(Storage &&storage, Value &&value) {
431 static constexpr std::array<std::size_t, 2> position{
432 storage_position<Columns>(Indexes...)};
433
434 using row = std::integral_constant<std::size_t, position[0]>;
435 using column = std::integral_constant<std::size_t, position[1]>;
436
437 if constexpr (requires {
438 requires std::is_lvalue_reference_v<
439 decltype(storage[row{}, column{}])>;
440 }) {
441 storage[row{}, column{}] = std::forward<Value>(value);
442 } else {
443 write_element<Columns>(storage, std::forward<Value>(value), Indexes...);
444 }
445}

577template <typename To, typename Tuple>
578constexpr auto find_first_convertible_index() -> std::size_t {
579 constexpr std::size_t size{std::tuple_size_v<Tuple>};
580
581 constexpr auto search{
582 []<std::size_t... Indexes>(std::index_sequence<Indexes...>) {
583 std::size_t result{0};
584
585 (void)((std::is_convertible_v<std::tuple_element_t<Indexes, Tuple>, To>
586 ? true
587 : (++result, false)) ||
588 ...);
589
590 return result;
591 }};
592
593 return search(std::make_index_sequence<size>{});
594}

722template <typename Lhs, typename Rhs, typename LhsUnderlying,
723 typename RhsUnderlying = LhsUnderlying>
724constexpr auto is_same_scale() -> bool {
725 if constexpr (std::same_as<Lhs, Rhs>) {
726 return true;
727 } else if constexpr (requires { typename std::common_type_t<Lhs, Rhs>; }) {
728 using lhs_caster = element_caster<Lhs, LhsUnderlying>;
729 using rhs_caster = element_caster<Rhs, RhsUnderlying>;
730
731 return lhs_caster{}(LhsUnderlying{1}) == rhs_caster{}(RhsUnderlying{1}) and
732 lhs_caster{}(LhsUnderlying{0}) == rhs_caster{}(RhsUnderlying{0});
733 } else {
734 return false;
735 }
736}

779template <typename From, typename To>
780constexpr auto keeps_fixed_elements() -> bool {
781 using from_matrix = std::remove_cvref_t<From>;
782 using to_matrix = std::remove_cvref_t<To>;
783 using storage = typename to_matrix::matrix;
784 using from_underlying = typename from_matrix::underlying;
785 using to_underlying = typename to_matrix::underlying;
786 bool result{true};
787
788 for_constexpr<to_matrix::rows>([&result](auto i) {
789 for_constexpr<to_matrix::columns>([&result, &i](auto j) {
790 if constexpr (not writable_element_at<storage, to_matrix::columns, i,
791 j>) {
792 using from_element = element_at<from_matrix, i, j>;
793 using to_element = element_at<to_matrix, i, j>;
794
795 constexpr to_underlying fixed{
796 read_element<to_matrix::columns>(storage{}, i, j)};
797
798 // Copy-initialization: the implicit conversion the concept verifies.
799 const to_element converted =
800 element_caster<from_element, from_underlying>{}(fixed);
801
802 result &=
803 element_caster<to_underlying, to_element>{}(converted) == fixed;
804 }
805 });
806 });
807
808 return result;
809}

831 decltype(std::declval<const From &>().data())> and

843template <typename To>
844[[nodiscard]] constexpr auto converted_storage(auto &&from) -> decltype(auto) {
845 using from_matrix = std::remove_cvref_t<decltype(from)>;
846 using to_matrix = std::remove_cvref_t<To>;
847
848 if constexpr (is_same_scale_typed_matrix<from_matrix, to_matrix>()) {
849 return std::forward<decltype(from)>(from).data();
850 } else {
851 using storage = typename to_matrix::matrix;
852 using from_underlying = typename from_matrix::underlying;
853 using to_underlying = typename to_matrix::underlying;
854
855 const storage evaluated{from.data()};
856 storage result{};
857
858 // The casters are named directly, the `cast` object is declared after the
859 // class.
860 for_constexpr<to_matrix::rows>([&](auto i) {
861 for_constexpr<to_matrix::columns>([&](auto j) {
