118// The call operators that compute a product, rather than name one, are
119// templates deducing their operands, so that an ill-formed product is a
120// substitution failure: `product` is then not a type, which a `requires`
121// expression detects, instead of a hard error.

124 [[nodiscard]] static constexpr auto operator()(const Factor &lhs,
125 const Rhs &rhs)

266template <std::size_t Size, typename... Indexes, std::size_t... Is>
267[[nodiscard]] consteval auto
268distinct_indexes([[maybe_unused]] std::index_sequence<Is...> indexes)
269 -> distinct_indexes_t<Size> {
270 const std::array<const void *, Size> tokens{
271 &type_token<index_types_at<Is, Indexes...>>...};
272 std::array<const void *, Size> distinct_tokens{};
273 distinct_indexes_t<Size> result;
274 auto distinct_token{distinct_tokens.begin()};
275 auto distinct_index{result.indexes.begin()};
276 std::size_t index{0};
277
278 for (const void *const token : tokens) {
279 bool distinct{true};
280
281 for (auto seen{distinct_tokens.begin()}; seen != distinct_token; ++seen) {
282 distinct = distinct and *seen != token;
283 }
284
285 if (distinct) {
286 *distinct_token = token;
287 ++distinct_token;
288 *distinct_index = index;
289 ++distinct_index;
290 ++result.count;
291 }
292
293 ++index;
294 }
295
296 return result;
297}

468template <std::size_t Columns>
469[[nodiscard]] constexpr auto
470storage_position(std::convertible_to<std::size_t> auto... indexes)
471 -> std::array<std::size_t, 2> {
472 if constexpr (sizeof...(indexes) == 2) {
473 return {static_cast<std::size_t>(indexes)...};
474 } else if constexpr (sizeof...(indexes) == 1) {
475 const std::size_t index{static_cast<std::size_t>(indexes)...};
476 return {index / Columns, index % Columns};
477 } else {
478 return {0, 0};
479 }
480}

494template <std::size_t Columns, typename Storage>
495[[nodiscard]] constexpr auto
496read_element(Storage &&storage,
497 std::convertible_to<std::size_t> auto... indexes)
498 -> decltype(auto) {
499 const auto [row, column]{storage_position<Columns>(indexes...)};
500
501 if constexpr (same_as_typed_matrix<Storage>) {
502 return read_element<Columns>(storage.data(), indexes...);
503 } else if constexpr (requires { access_element(storage, row, column); }) {
504 using storage_index = storage_index_t<std::remove_reference_t<Storage>>;
505
506 return access_element(storage, static_cast<storage_index>(row),
507 static_cast<storage_index>(column));
508 } else if constexpr (requires { storage.eval(); }) {
509 // `auto{...}` copies the element out of the local evaluation before it
510 // goes out of scope.
511 auto evaluated{storage.eval()};
512 return auto{read_element<Columns>(evaluated, indexes...)};
513 } else {
514 static_assert(sizeof(Storage) == 0,
515 "Element access is not supported for this linear algebra "
516 "backend.");
517 }
518}

533template <std::size_t Columns, auto... Indexes, typename Storage>
534[[nodiscard]] constexpr auto read_element_at(Storage &&storage)
535 -> decltype(auto) {
536 static constexpr std::array<std::size_t, 2> position{
537 storage_position<Columns>(Indexes...)};
538
539 using row = std::integral_constant<std::size_t, position[0]>;
540 using column = std::integral_constant<std::size_t, position[1]>;
541
542 if constexpr (same_as_typed_matrix<Storage>) {
543 return read_element_at<Columns, Indexes...>(storage.data());
544 } else if constexpr (requires { storage[row{}, column{}]; }) {
545 return storage[row{}, column{}];
546 } else {
547 return read_element<Columns>(storage, Indexes...);
548 }
549}

555template <std::size_t Columns, typename Storage, typename Value>
556constexpr void write_element(Storage &&storage, Value &&value,
557 std::convertible_to<std::size_t> auto... indexes) {
558 static_assert(std::is_lvalue_reference_v<decltype(read_element<Columns>(
559 storage, indexes...))>,
560 "Element write access requires a referenceable storage "
561 "element.");

563 read_element<Columns>(storage, indexes...) = std::forward<Value>(value);

594template <std::size_t Columns, auto... Indexes, typename Storage,
595 typename Value>
596constexpr void write_element_at(Storage &&storage, Value &&value) {
597 static constexpr std::array<std::size_t, 2> position{
598 storage_position<Columns>(Indexes...)};
599
600 using row = std::integral_constant<std::size_t, position[0]>;
601 using column = std::integral_constant<std::size_t, position[1]>;
602
603 if constexpr (requires {
604 requires std::is_lvalue_reference_v<
605 decltype(storage[row{}, column{}])>;
606 }) {
607 storage[row{}, column{}] = std::forward<Value>(value);
608 } else {
609 write_element<Columns>(storage, std::forward<Value>(value), Indexes...);
610 }
611}

743template <typename To, typename Tuple>
744constexpr auto find_first_convertible_index() -> std::size_t {
745 constexpr std::size_t size{std::tuple_size_v<Tuple>};
746
747 constexpr auto search{
748 []<std::size_t... Indexes>(std::index_sequence<Indexes...>) {
749 std::size_t result{0};
750
751 (void)((std::is_convertible_v<std::tuple_element_t<Indexes, Tuple>, To>
752 ? true
753 : (++result, false)) ||
754 ...);
755
756 return result;
757 }};
758
759 return search(std::make_index_sequence<size>{});
760}

793template <typename Type> constexpr auto is_uniform_typed_matrix() -> bool {
794 using matrix = std::remove_cvref_t<Type>;
795 bool result{true};

797 for_constexpr<matrix::rows>([&result](auto i) {
798 for_constexpr<matrix::columns>([&result, &i](auto j) {

801 });
