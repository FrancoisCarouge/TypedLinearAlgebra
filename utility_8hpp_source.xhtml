292template <std::size_t Columns>
293[[nodiscard]] constexpr auto
294storage_position(std::convertible_to<std::size_t> auto... indexes)
295 -> std::array<std::size_t, 2> {
296 if constexpr (sizeof...(indexes) == 2) {
297 return {static_cast<std::size_t>(indexes)...};
298 } else if constexpr (sizeof...(indexes) == 1) {
299 const std::size_t index{static_cast<std::size_t>(indexes)...};
300 return {index / Columns, index % Columns};
301 } else {
302 return {0, 0};
303 }
304}

318template <std::size_t Columns, typename Storage>
319[[nodiscard]] constexpr auto
320read_element(Storage &&storage,
321 std::convertible_to<std::size_t> auto... indexes)
322 -> decltype(auto) {
323 const auto [row, column]{storage_position<Columns>(indexes...)};
324
325 if constexpr (same_as_typed_matrix<Storage>) {
326 return read_element<Columns>(storage.data(), indexes...);
327 } else if constexpr (requires { access_element(storage, row, column); }) {
328 using storage_index = storage_index_t<std::remove_reference_t<Storage>>;
329
330 return access_element(storage, static_cast<storage_index>(row),
331 static_cast<storage_index>(column));
332 } else if constexpr (requires { storage.eval(); }) {
333 // `auto{...}` copies the element out of the local evaluation before it
334 // goes out of scope.
335 auto evaluated{storage.eval()};
336 return auto{read_element<Columns>(evaluated, indexes...)};
337 } else {
338 static_assert(sizeof(Storage) == 0,
339 "Element access is not supported for this linear algebra "
340 "backend.");
341 }
342}

357template <std::size_t Columns, auto... Indexes, typename Storage>
358[[nodiscard]] constexpr auto read_element_at(Storage &&storage)
359 -> decltype(auto) {
360 static constexpr std::array<std::size_t, 2> position{
361 storage_position<Columns>(Indexes...)};
362
363 using row = std::integral_constant<std::size_t, position[0]>;
364 using column = std::integral_constant<std::size_t, position[1]>;
365
366 if constexpr (same_as_typed_matrix<Storage>) {
367 return read_element_at<Columns, Indexes...>(storage.data());
368 } else if constexpr (requires { storage[row{}, column{}]; }) {
369 return storage[row{}, column{}];
370 } else {
371 return read_element<Columns>(storage, Indexes...);
372 }
373}

379template <std::size_t Columns, typename Storage, typename Value>
380constexpr void write_element(Storage &&storage, Value &&value,
381 std::convertible_to<std::size_t> auto... indexes) {
382 static_assert(std::is_lvalue_reference_v<decltype(read_element<Columns>(
383 storage, indexes...))>,
384 "Element write access requires a referenceable storage "
385 "element.");
386
387 read_element<Columns>(storage, indexes...) = std::forward<Value>(value);

397template <typename Storage, std::size_t Columns, std::size_t Row,
398 std::size_t Column>
399concept writable_element_at =
400 requires(Storage &storage) {
401 requires std::is_lvalue_reference_v<
402 decltype(storage[std::integral_constant<std::size_t, Row>{},
403 std::integral_constant<std::size_t, Column>{}])>;
404 } or
405 std::is_lvalue_reference_v<decltype(read_element<Columns>(
406 std::declval<Storage &>(), Row, Column))>;

418template <std::size_t Columns, auto... Indexes, typename Storage,
419 typename Value>
420constexpr void write_element_at(Storage &&storage, Value &&value) {
421 static constexpr std::array<std::size_t, 2> position{
422 storage_position<Columns>(Indexes...)};
423
424 using row = std::integral_constant<std::size_t, position[0]>;
425 using column = std::integral_constant<std::size_t, position[1]>;
426
427 if constexpr (requires {
428 requires std::is_lvalue_reference_v<
429 decltype(storage[row{}, column{}])>;
430 }) {
431 storage[row{}, column{}] = std::forward<Value>(value);
432 } else {
433 write_element<Columns>(storage, std::forward<Value>(value), Indexes...);
434 }
435}

567template <typename To, typename Tuple>
568constexpr auto find_first_convertible_index() -> std::size_t {
569 constexpr std::size_t size{std::tuple_size_v<Tuple>};
570
571 constexpr auto search{
572 []<std::size_t... Indexes>(std::index_sequence<Indexes...>) {
573 std::size_t result{0};
574
575 (void)((std::is_convertible_v<std::tuple_element_t<Indexes, Tuple>, To>
576 ? true
577 : (++result, false)) ||
578 ...);
579
580 return result;
581 }};
582
583 return search(std::make_index_sequence<size>{});
584}

712template <typename Lhs, typename Rhs, typename LhsUnderlying,
713 typename RhsUnderlying = LhsUnderlying>
714constexpr auto is_same_scale() -> bool {
715 if constexpr (std::same_as<Lhs, Rhs>) {
716 return true;
717 } else if constexpr (requires { typename std::common_type_t<Lhs, Rhs>; }) {
718 using lhs_caster = element_caster<Lhs, LhsUnderlying>;
719 using rhs_caster = element_caster<Rhs, RhsUnderlying>;
720
721 return lhs_caster{}(LhsUnderlying{1}) == rhs_caster{}(RhsUnderlying{1}) and
722 lhs_caster{}(LhsUnderlying{0}) == rhs_caster{}(RhsUnderlying{0});
723 } else {
724 return false;
725 }
726}

769template <typename From, typename To>
770constexpr auto keeps_fixed_elements() -> bool {
771 using from_matrix = std::remove_cvref_t<From>;
772 using to_matrix = std::remove_cvref_t<To>;
773 using storage = typename to_matrix::matrix;
774 using from_underlying = typename from_matrix::underlying;
775 using to_underlying = typename to_matrix::underlying;
776 bool result{true};
777
778 for_constexpr<to_matrix::rows>([&result](auto i) {
779 for_constexpr<to_matrix::columns>([&result, &i](auto j) {
780 if constexpr (not writable_element_at<storage, to_matrix::columns, i,
781 j>) {
782 using from_element = element_at<from_matrix, i, j>;
783 using to_element = element_at<to_matrix, i, j>;
784
785 constexpr to_underlying fixed{
786 read_element<to_matrix::columns>(storage{}, i, j)};
787
788 // Copy-initialization: the implicit conversion the concept verifies.
789 const to_element converted =
790 element_caster<from_element, from_underlying>{}(fixed);
791
792 result &=
793 element_caster<to_underlying, to_element>{}(converted) == fixed;
794 }
795 });
796 });
797
798 return result;
799}

821 decltype(std::declval<const From &>().data())> and

833template <typename To>
834[[nodiscard]] constexpr auto converted_storage(auto &&from) -> decltype(auto) {
835 using from_matrix = std::remove_cvref_t<decltype(from)>;
836 using to_matrix = std::remove_cvref_t<To>;
837
838 if constexpr (is_same_scale_typed_matrix<from_matrix, to_matrix>()) {
839 return std::forward<decltype(from)>(from).data();
840 } else {
841 using storage = typename to_matrix::matrix;
842 using from_underlying = typename from_matrix::underlying;
843 using to_underlying = typename to_matrix::underlying;
844
845 const storage evaluated{from.data()};
846 storage result{};
847
848 // The casters are named directly, the `cast` object is declared after the
849 // class.
850 for_constexpr<to_matrix::rows>([&](auto i) {
851 for_constexpr<to_matrix::columns>([&](auto j) {
852 using from_element = element_at<from_matrix, i, j>;
853 using to_element = element_at<to_matrix, i, j>;
854
855 if constexpr (writable_element_at<storage, to_matrix::columns, i, j>) {
856 // Copy-initialization: the implicit conversion the concept verified.
857 const to_element converted =
858 element_caster<from_element, from_underlying>{}(
