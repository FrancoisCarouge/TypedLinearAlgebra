214template <std::size_t Columns, typename Storage>
215[[nodiscard]] constexpr auto
216read_element(Storage &&storage,
217 std::convertible_to<std::size_t> auto... indexes)
218 -> decltype(auto) {
219 const auto [row, column]{[&] -> std::array<std::size_t, 2> {
220 if constexpr (sizeof...(indexes) == 2) {
221 return {static_cast<std::size_t>(indexes)...};
222 } else if constexpr (sizeof...(indexes) == 1) {
223 const std::size_t index{static_cast<std::size_t>(indexes)...};
224 return {index / Columns, index % Columns};
225 } else {
226 return {0, 0};
227 }
228 }()};
229
230 if constexpr (same_as_typed_matrix<Storage>) {
231 return read_element<Columns>(storage.data(), indexes...);
232 } else if constexpr (requires { storage[row, column]; }) {
233 return storage[row, column];
234 } else if constexpr (requires { storage(row, column); }) {
235 return storage(row, column);
236 } else if constexpr (requires { storage.at(row, column); }) {
237 return storage.at(row, column);
238 } else if constexpr (requires { storage.eval(); }) {
239 // `auto(...)` copies the element out of the local evaluation before it
240 // goes out of scope.
241 auto evaluated{storage.eval()};
242 return auto(read_element<Columns>(evaluated, indexes...));
243 } else {
244 static_assert(sizeof(Storage) == 0,
245 "Element access is not supported for this linear algebra "
246 "backend.");
247 }
248}

254template <std::size_t Columns, typename Storage, typename Value>
255constexpr void write_element(Storage &&storage, Value &&value,
256 std::convertible_to<std::size_t> auto... indexes) {
257 static_assert(std::is_lvalue_reference_v<decltype(read_element<Columns>(
258 storage, indexes...))>,
259 "Element write access requires a referenceable storage "
260 "element.");

262 read_element<Columns>(storage, indexes...) = std::forward<Value>(value);

396template <typename To, typename Tuple>
397constexpr auto find_first_convertible_index() -> std::size_t {
398 constexpr std::size_t size{std::tuple_size_v<Tuple>};
399
400 constexpr auto search{
401 []<std::size_t... Indexes>(std::index_sequence<Indexes...>) {
402 std::size_t result{size};
403
404 (void)((std::is_convertible_v<std::tuple_element_t<Indexes, Tuple>, To>
405 ? (result = Indexes, true)
406 : false) ||
407 ...);
408
409 return result;
410 }};
411
412 return search(std::make_index_sequence<size>{});
413}

432template <typename Type> constexpr auto is_uniform_typed_matrix() -> bool {
433 using matrix = std::remove_cvref_t<Type>;
434 bool result{true};
435
436 for_constexpr<matrix::rows>([&result](auto i) {
437 for_constexpr<matrix::columns>([&result, &i](auto j) {
438 result &=
439 std::is_same_v<element_at<matrix, i, j>, element_at<matrix, 0, 0>>;
440 });
441 });
442
443 return result;
444}

568template <typename Type> constexpr auto is_distinct_typed_matrix() -> bool {
569 using matrix = std::remove_cvref_t<Type>;
570 bool result{true};
571
572 for_constexpr<matrix::rows>([&result](auto i) {
573 for_constexpr<matrix::columns>([&result, &i](auto j) {
574 for_constexpr<matrix::rows>([&result, &i, &j](auto k) {
575 for_constexpr<matrix::columns>([&result, &i, &j, &k](auto l) {
576 if constexpr (i != k || j != l) {
577 using lhs = element_at<matrix, i, j>;
578 using rhs = element_at<matrix, k, l>;
579
580 result &= not(std::is_convertible_v<lhs, rhs> or
581 std::is_convertible_v<rhs, lhs> or
582 requires { typename std::common_type_t<lhs, rhs>; });
583 }
584 });
585 });
586 });
587 });
588
589 return result;
590}
