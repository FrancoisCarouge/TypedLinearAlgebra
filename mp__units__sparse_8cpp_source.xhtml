49template <auto QuantityReference>
50using quantity = mp_units::quantity<QuantityReference, representation>;
51
52using mp_units::one;
53using mp_units::si::unit_symbols::m;
54using mp_units::si::unit_symbols::s;
55using literals::operator""_o;
56
57using position = quantity<mp_units::isq::length[m]>;
58using velocity = quantity<mp_units::isq::speed[m / s]>;
59using inverse_position = quantity<one / mp_units::isq::length[m]>;
60using inverse_velocity = quantity<one / mp_units::isq::speed[m / s]>;
61
63[[maybe_unused]] const auto sample{[] -> int {
64 // A state transition, a linear map: the column indexes are reciprocal to the
65 // state's. The pattern is part of the type: the bottom left element is a
66 // structural zero, without storage.
67 using transition =
68 typed_sparse_matrix<representation, sparse::upper_triangular<2>,
69 std::tuple<position, velocity>,
70 std::tuple<inverse_position, inverse_velocity>>;
71
72 static_assert(sizeof(transition) == 3 * sizeof(representation));
73
74 // Each stored element is written with its own type: a ratio, a duration.
75 transition model;
76
77 model.at<0, 0>(1. * one);
78 model.at<0, 1>(2. * s);
79 model.at<1, 1>(1. * one);
80
81 // A structural zero is read with its own type, an inverse duration, by value.
82 // Writing it does not compile: `model.at<1, 0>(1. / s);`.
83 assert((model.at<1, 0>() == 0. / s));
84
85 // The product only computes the terms the patterns admit: three of the four
86 // of a dense product of these shapes.
87 using state = typed_sparse_column_vector<representation, sparse::dense<2, 1>,
88 position, velocity>;
89
90 const state initial{1. * m, 3. * m / s};
91 const state predicted{model * initial};
92
93 assert(predicted.at<0>() == 7. * m);
94 assert(predicted.at<1>() == 3. * m / s);
95
96 // A pattern of any shape is a list of positions.
97 constexpr sparse::pattern ends{sparse::shape<3, 1>, {{0, 0}, {2, 0}}};
98
99 using sparse_state = typed_sparse_column_vector<representation, ends,
100 position, velocity, position>;
101
102 static_assert(sizeof(sparse_state) == 2 * sizeof(representation));
103
104 // Each operation computes the pattern of its result at compile time.
105 const sparse_state first{
106 sparse::make_matrix<sparse_state>(1. * m, 0_o, 2. * m)};
107 const auto sum{first + first};
108
109 static_assert(decltype(sum)::matrix::structure == ends);
110 assert(sum.at<2>() == 4. * m);
111
112 return 0;
113}()};
