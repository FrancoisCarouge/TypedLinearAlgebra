42using singleton = matrix<double, 1, 1>;
43using row = matrix<double, 1, 3>;
44using column = matrix<double, 3, 1>;
45using plane = matrix<double, 2, 3>;
46
47// Positive: a typed matrix always has the same shape as itself, the relation
48// is reflexive over every rank.
49static_assert(same_shape<singleton, singleton>);
50static_assert(same_shape<row, row>);
51static_assert(same_shape<column, column>);
52static_assert(same_shape<plane, plane>);
53
54// Positive: the shape is the row and column counts only. The element types,
55// and the backend element storage, play no part; two matrices of different
56// element types but identical dimensions have the same shape.
57static_assert(same_shape<matrix<double, 2, 3>, matrix<float, 2, 3>>);
58static_assert(same_shape<matrix<double, 2, 3>, matrix<int, 2, 3>>);
59static_assert(same_shape<row_vector<double, 4>, row_vector<float, 4>>);
60
61// Positive: the relation is symmetric in its two arguments.
62static_assert(same_shape<matrix<double, 2, 3>, matrix<float, 2, 3>> ==
63 same_shape<matrix<float, 2, 3>, matrix<double, 2, 3>>);
64static_assert(same_shape<row, column> == same_shape<column, row>);
65
66// Positive, special case: cv-qualification and references are stripped with
67// `std::remove_cvref_t` before the dimensions are compared.
68static_assert(same_shape<const plane, plane>);
69static_assert(same_shape<plane &, const plane &&>);
70static_assert(same_shape<const plane &, volatile plane>);
71static_assert(same_shape<const volatile plane &, plane>);
72
73// Negative: a differing row count is a different shape.
74static_assert(not same_shape<plane, matrix<double, 3, 3>>);
75static_assert(not same_shape<singleton, column>);
76
77// Negative: a differing column count is a different shape.
78static_assert(not same_shape<plane, matrix<double, 2, 4>>);
79static_assert(not same_shape<singleton, row>);
80
81// Negative: a matrix and its transpose shape are not the same shape, even
82// though they hold the same number of elements.
83static_assert(not same_shape<plane, matrix<double, 3, 2>>);
84static_assert(not same_shape<row, column>);
85
86// Negative: a row vector and the column vector of the same length are both
87// rank one with the same element count, but not the same shape.
88static_assert(not same_shape<row_vector<double, 4>, column_vector<double, 4>>);
89
90// Negative, special case: both arguments must be typed matrices. Any other
91// type makes the concept unsatisfied through a substitution failure in the
92// immediate context, never a hard error on a missing `::rows`/`::columns`.
93static_assert(not same_shape<plane, double>);
94static_assert(not same_shape<double, plane>);
95static_assert(not same_shape<plane, void>);
96static_assert(not same_shape<plane, std::tuple<double>>);
97static_assert(not same_shape<int, float>);
98
99// Negative, special case: the underlying backend matrix, and a pointer to a
100// typed matrix, are not typed matrices themselves.
101static_assert(not same_shape<plane, eigen::matrix<double, 2, 3>>);
102static_assert(not same_shape<eigen::matrix<double, 2, 3>, plane>);
103static_assert(not same_shape<plane, plane *>);
104
105// Negative, special case: an incomplete type is probed safely, the concept is
106// simply unsatisfied.
107struct incomplete;
108static_assert(not same_shape<plane, incomplete>);
109
110// Agrees with the shape-carrying member counts and the shape concepts.
111static_assert(same_shape<row, matrix<double, row::rows, row::columns>>);
112static_assert(same_shape<row, row> and row_typed_matrix<row> and
113 not column_typed_matrix<row>);
114static_assert(same_shape<singleton, singleton> and
115 rank_typed_matrix<singleton, 0>);
116static_assert(not same_shape<row, column> and rank_typed_matrix<row, 1> and
117 rank_typed_matrix<column, 1>);
