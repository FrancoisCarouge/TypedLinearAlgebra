68[[maybe_unused]] const auto sample{[] -> int {
69 using durations = column_vector<seconds, minutes, hours>;
70
71 std::vector v0(std::size_t{3}, representation{});
72 std::mdspan s0{v0.data(), column_extents<3>{}};
73 durations x0{s0};
74
75 // Element assignment keeps each period.
76 x0.at<0>(seconds{30.});
77 x0.at<1>(minutes{2.});
