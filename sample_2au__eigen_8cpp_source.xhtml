74[[maybe_unused]] const auto sample{[] -> int {
75 using au::symbols::m;
76 using au::symbols::s;
77
78 constexpr auto m2{au::squared(m)};
79 constexpr auto s2{au::squared(s)};
80
81 // Set up a heterogenous column vector type for the sample.
82 using state = column_vector<position, velocity, acceleration>;
83
84 // A vector of quantities:
85 state x0{3. * m, 2. * m / s, 1. * m / s2};
86
87 // Printable.
88 std::println("x0 = {}", x0);
89 assert(std::format("{}", x0) == "[[3 m], [2 m / s], [1 m / s^2]]");
90
91 // Element assignment and access.
92 x0.at<1>(2.5 * m / s);
93 auto x0_1{x0.at<1>()};
94 assert(x0_1 == 2.5 * m / s);
95 assert(std::format("{}", x0_1) == "2.5 m / s");
96
97 // Multiplication with a scalar factor.
98 state x1{x0 * 3.};
99 assert(std::format("{}", x1) == "[[9 m], [7.5 m / s], [3 m / s^2]]");
100
101 // Division with a scalar divisor.
102 state x2{x1 / 2.};
103 assert(std::format("{}", x2) == "[[4.5 m], [3.75 m / s], [1.5 m / s^2]]");
104
105 // Substraction of two vectors of the same types.
106 state x3{x2 - x0};
107 assert(std::format("{}", x3) == "[[1.5 m], [1.25 m / s], [0.5 m / s^2]]");
108
109 // Additions of two vectors of the same types.
110 state x4{x3 + x3};
111 assert(std::format("{}", x4) == "[[3 m], [2.5 m / s], [1 m / s^2]]");
112
113 // Matrix and uniform typed access.
114 using position_2d_uncertainty =
115 matrix<std::tuple<position, position>, std::tuple<position, position>>;
116 position_2d_uncertainty p0;
117
118 p0.at<0, 1>(9. * m2);
119 assert((p0[0, 1] == 9. * m2));
120
121 p0.at<0, 1>(16. * m2);
122 assert((p0(0, 1) == 16. * m2));
123
124 // Matrix multiplication: column vector by row vector outer product,
125 // producing an area typed matrix.
126 const column_vector<position, position> c{1. * m, 2. * m};
127 const row_vector<position, position> r{3. * m, 4. * m};
128 const matrix<std::tuple<position, position>, std::tuple<position, position>>
129 cr{c * r};
130
131 assert((cr.at<0, 0>() == 3. * m2));
132 assert((cr.at<0, 1>() == 4. * m2));
133 assert((cr.at<1, 0>() == 6. * m2));
134 assert((cr.at<1, 1>() == 8. * m2));
135
136 return 0;
137}()};
