71// Just like Eigen, the fcarouge::typed_matrix
72// arithmetic operators return lazy expression templates; store their evaluated
73// concrete type (`PlainObject`) in a quantity instead. Concrete
74// matrices/vectors map to themselves.
75//
76// The `typename T::PlainObject` requirement is checked first and short-circuits
77// the rest of the constraint: `representation_canonical_type` is instantiated
78// for every representation type (including `int`, `double`, ...), and
79// `Eigen::EigenBase<T>` is ill-formed for a non-Eigen `T`, so it must not be
80// instantiated unless `T` already looks like an Eigen type.
81template <typename Matrix, typename RowIndexes, typename ColumnIndexes>
82 requires requires { typename Matrix::PlainObject; } &&
83 std::derived_from<Matrix, Eigen::EigenBase<Matrix>>
84struct mp_units::representation_canonical_type<
85 fcarouge::typed_matrix<Matrix, RowIndexes, ColumnIndexes>> {
86 using type =
87 fcarouge::typed_matrix<std::remove_cvref_t<typename Matrix::PlainObject>,
88 RowIndexes, ColumnIndexes>;
89};
90
91// Just like Blaze, the fcarouge::typed_matrix
92// does not expose the `value_type`/`element_type` names the library detects
93// automatically, so map its `ElementType` explicitly.
94template <typename Matrix, typename RowIndexes, typename ColumnIndexes>
95 requires requires { typename Matrix::PlainObject; } &&
96 std::derived_from<Matrix, Eigen::EigenBase<Matrix>>
97struct mp_units::representation_underlying_type<
98 fcarouge::typed_matrix<Matrix, RowIndexes, ColumnIndexes>> {
99 using type =
100 fcarouge::typed_matrix<Matrix, RowIndexes, ColumnIndexes>::underlying;
101};
