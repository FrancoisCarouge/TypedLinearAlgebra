62template <typename Type = double, std::size_t Row = 1, std::size_t Column = 1>
63struct matrix {
65 std::array<Type, Row * Column> elements{};
66
68 [[nodiscard]] constexpr auto operator[](this auto &&self, std::size_t row,
69 std::size_t column) -> auto & {
70 return self.elements[(row * Column) + column];
71 }
72
74 [[nodiscard]] friend constexpr auto operator+(const matrix &lhs,
75 const matrix &rhs) -> matrix {
76 return transform(std::plus{}, lhs, rhs);
77 }
78
80 [[nodiscard]] friend constexpr auto operator-(const matrix &lhs,
81 const matrix &rhs) -> matrix {
82 return transform(std::minus{}, lhs, rhs);
83 }
84
86 [[nodiscard]] friend constexpr auto operator-(const matrix &value) -> matrix {
87 return transform(std::negate{}, value);
88 }
89
91 [[nodiscard]] friend constexpr auto operator*(const matrix &lhs,
92 const Type &rhs) -> matrix {
93 return transform(
94 [&rhs](const Type &element) -> Type { return element * rhs; }, lhs);
95 }
96
98 [[nodiscard]] friend constexpr auto operator*(const Type &lhs,
99 const matrix &rhs) -> matrix {
100 return transform(
101 [&lhs](const Type &element) -> Type { return lhs * element; }, rhs);
102 }
103
105 template <std::size_t Other>
106 [[nodiscard]] friend constexpr auto
107 operator*(const matrix &lhs, const matrix<Type, Column, Other> &rhs)
108 -> matrix<Type, Row, Other> {
109 matrix<Type, Row, Other> result;
110
111 for (std::size_t i{0}; i < Row; ++i) {
112 for (std::size_t j{0}; j < Other; ++j) {
113 for (std::size_t k{0}; k < Column; ++k) {
114 result[i, j] += lhs[i, k] * rhs[k, j];
115 }
116 }
117 }
118
119 return result;
120 }
121
122private:
125 template <std::size_t Position>
126 [[nodiscard]] static constexpr auto transform_at(const auto &function,
127 const auto &...values)
128 -> Type {
129 return function(std::get<Position>(values.elements)...);
130 }
131
134 [[nodiscard]] static constexpr auto transform(const auto &function,
135 const auto &...values)
136 -> matrix {
137 return [&]<std::size_t... Positions>(
138 std::index_sequence<Positions...>) -> matrix {
139 return {{transform_at<Positions>(function, values...)...}};
140 }(std::make_index_sequence<Row * Column>{});
141 }
142};

145} // namespace fcarouge::freestanding
