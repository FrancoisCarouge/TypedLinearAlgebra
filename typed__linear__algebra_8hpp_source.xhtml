390 //
391 // Defined in the class: MSVC drops the value-dependent explicit specifier
392 // of a constructor template once its out-of-class definition is seen, see
393 // https://developercommunity.visualstudio.com/t/856958.
394 template <other_tuple_like_vector<typed_matrix> Value>
395 constexpr explicit(not tla::is_convertible_tuple_like<Value, typed_matrix>())
396 typed_matrix(const Value &value) {
397 tla::for_constexpr<rows * columns>([&](auto i) {
398 // Convert to the element type first, e.g. to the element's unit, so the
399 // caster only ever stores values of the element type. The caster is
400 // named directly, the `cast` object is declared after the class.
401 const element<i> converted(get<i>(value));
402 tla::write_element<columns>(
403 storage, element_caster<underlying, element<i>>{}(converted), i);
404 });
405 }
