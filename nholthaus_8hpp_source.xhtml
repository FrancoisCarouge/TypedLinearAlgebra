49// True for a nholthaus/units quantity, regardless of its conversion factor,
50// representation, or numerical scale. `is_unit_v` already excludes plain
51// arithmetic types.
52template <typename T>
53concept nholthaus_quantity = units::traits::is_unit_v<std::remove_cvref_t<T>>;
54
55// The arithmetic storage type backing a nholthaus/units quantity (e.g.
56// `double` for `units::length::meters<double>`).
57template <nholthaus_quantity Quantity>
58using nholthaus_representation = typename units::traits::unit_traits<
59 std::remove_cvref_t<Quantity>>::underlying_type;
60
61// Teach the typed linear algebra library how to convert underlying scalar
62// types to and from nholthaus/units' quantity type.
63template <typename To, nholthaus_quantity From>
64struct element_caster<To, From> {
65 [[nodiscard]] static constexpr auto operator()(From value) -> To {
66 static_assert(
67 std::same_as<nholthaus_representation<From>, std::remove_cvref_t<To>>,
68 "The underlying storage type must be identical to the "
69 "quantity representation type to guarantee the conversion "
70 "is explicitly decided by the end-user.");
71
72 return value.value();
73 }
74};
75
76template <nholthaus_quantity To, typename From>
77struct element_caster<To, From> {
78 [[nodiscard]] static constexpr auto operator()(From value) -> To {
79 static_assert(
80 std::same_as<nholthaus_representation<To>, std::remove_cvref_t<From>>,
81 "The underlying storage type must be identical to the "
82 "quantity representation type to guarantee the conversion "
83 "is explicitly decided by the end-user.");
84
85 return To(value);
86 }
87};
88
89template <nholthaus_quantity To, typename From>
90struct element_caster<To &, From &> {
91 // A quantity reference cannot be safely materialized out of a
92 // representation reference. It would be undefined behavior even if the
93 // size, padding, alignment, aliasing are controlled. Therefore the best we
94 // can do is to return a constant quantity value to inform the end-user
95 // lvalue reference assignment cannot be supported.
96 [[nodiscard]] static constexpr auto operator()(From value) -> const To {
97 static_assert(
98 std::same_as<nholthaus_representation<To>, std::remove_cvref_t<From>>,
99 "The underlying storage type must be identical to the "
100 "quantity representation type to guarantee the conversion "
101 "is explicitly decided by the end-user.");
102
103 return To(value);
104 }
105};
